$$

$$

$$

$$-\frac{201}{8}+\frac{3}{2}v+\frac{3}{4}w +\frac{2}{3}v-\frac{1}{3}w=-9$$
$$-\frac{201}{8}+\left( \frac{3}{2}v +\frac{2}{3}v  \right)+ \left( \frac{3}{4}w -\frac{1}{3}w \right)=-9$$

$$\frac{13}{6}v+\frac{5}{12}w =-9 +\frac{201}{8}$$

$$v=\frac{6}{13}\left( \frac{129}{8} -\frac{5}{12}w \right)$$
$$v=\frac{387}{52}-\frac{5}{26}w$$

$$\to7v-\frac{1}{3}w=\frac{27}{4}$$

$$-\frac{131}{78}w=-\frac{1179}{26}$$
$$
w= \frac{-\frac{1179}{26}}{-\frac{131}{78}}
$$
$$w=27$$
$$
v=\frac{387}{52}-\frac{5}{26}w
$$
$$v=\frac{387}{52}-\frac{5}{26}(27)$$
$$v=\frac{9}{4}$$
$$
u=\frac{67-4v-2w}{8}

\end{cases}
$$

$$

$$
$$

$$b=\frac{3}{5}c \tag{2}$$

$$\to 5a+2b-7c=-24$$
$$5(9-b-c)+2b-7c=-24$$
$$(45-5b-5c)+2b-7c=-24$$
$$(-5b+2b)+(-5c-7c)=-24-45$$
$$-3b-12c=-69$$
$$b=\frac{3}{5}c \tag{2}$$
$$-3\left( \frac{3}{5}c \right)-12c=-69$$
$$-\frac{69}{5}c=-69$$
$$c=\frac{-69}{-\frac{69}{5}}$$
